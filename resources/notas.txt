Here's the definitive list — 6 mnemonics, nothing else:
#	Pattern	Mnemonic	    Semantics
1	00E0	CLS	            Clear the display buffer
2	1nnn	JP nnn          PC = nnn (no +2)
3	6xkk	LD Vx, kk	    Vx = kk
4	7xkk	ADD Vx, kk	    Vx += kk (no carry flag)
5	Annn	LD I, nnn	    I = nnn
6	Dxyn	DRW Vx, Vy, n	    XOR 8×n sprite from I at (Vx,Vy); VF=1 on collision

opcode = 0xD01F
         ^ ^ ^
         | | +-- n   = opcode & 0x000F   (height, here 0xF = 15)
         | +---- y   = (opcode >> 4) & 0xF  (Vy, here 1)
         +------ x   = (opcode >> 8) & 0xF  (Vx, here 0)
nnn  = opcode & 0x0FFF
kk   = opcode & 0x00FF
